// Visual QA harness. Copied to test/_visual_snapshot_test.dart by the
// visual-qa skill, run once, then deleted. Never commit it under test/.
//
// Renders the real table screens with real fonts at several window sizes and
// writes PNGs to --dart-define=OUT=<dir>. Add cases at the bottom as needed.
import 'dart:io';
import 'dart:typed_data';
import 'dart:ui' as ui;

import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter/services.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:provider/provider.dart';
import 'package:poker_dice_dash/models/game_state.dart';
import 'package:poker_dice_dash/providers/blackjack_provider.dart';
import 'package:poker_dice_dash/providers/game_provider.dart';
import 'package:poker_dice_dash/services/p2p_service.dart';
import 'package:poker_dice_dash/ui/screens/blackjack_table_screen.dart';
import 'package:poker_dice_dash/ui/screens/game_table_screen.dart';

const outDir = String.fromEnvironment('OUT');
const fontsDir = String.fromEnvironment('FONTS');

// Phone landscape (the tightest), tablet and PC.
const sizes = [Size(800, 360), Size(640, 360), Size(1280, 720)];

Future<void> _loadFonts() async {
  final roboto = FontLoader('Roboto');
  for (final f in ['roboto-regular.ttf', 'roboto-bold.ttf', 'roboto-medium.ttf']) {
    final file = File('$fontsDir/$f');
    if (file.existsSync()) roboto.addFont(Future.value(ByteData.sublistView(file.readAsBytesSync())));
  }
  await roboto.load();
  // The logo title font (bundled in assets/fonts), otherwise it renders as blocks.
  final alfa = FontLoader('AlfaSlabOne')
    ..addFont(Future.value(ByteData.sublistView(File('assets/fonts/AlfaSlabOne-Regular.ttf').readAsBytesSync())));
  await alfa.load();
  final icons = FontLoader('MaterialIcons')
    ..addFont(Future.value(ByteData.sublistView(File('$fontsDir/materialicons-regular.otf').readAsBytesSync())));
  await icons.load();
}

Future<void> _snap(WidgetTester tester, GlobalKey key, String name) async {
  await tester.runAsync(() async {
    final boundary = key.currentContext!.findRenderObject() as RenderRepaintBoundary;
    final image = await boundary.toImage(pixelRatio: 1.5);
    final bytes = await image.toByteData(format: ui.ImageByteFormat.png);
    File('$outDir/$name.png').writeAsBytesSync(bytes!.buffer.asUint8List());
  });
}

Widget _app(GlobalKey key, Widget child) => RepaintBoundary(
      key: key,
      child: MaterialApp(
        debugShowCheckedModeBanner: false,
        theme: ThemeData(fontFamily: 'Roboto'),
        home: child,
      ),
    );

Future<void> _settle(WidgetTester tester, {int ticks = 40}) async {
  for (var i = 0; i < ticks; i++) {
    await tester.pump(const Duration(milliseconds: 100));
  }
  // Let image assets decode.
  await tester.runAsync(() => Future.delayed(const Duration(milliseconds: 300)));
  await tester.pump();
}

Future<void> _teardown(WidgetTester tester) async {
  await tester.pumpWidget(const SizedBox());
  await tester.pump(const Duration(seconds: 30)); // drain bot/animation timers
}

void main() {
  setUpAll(_loadFonts);

  for (final size in sizes) {
    final tag = '${size.width.toInt()}x${size.height.toInt()}';

    for (final mode in [GameMode.texasHoldem, GameMode.diceDash]) {
      for (final bots in [3, 5, 7]) {
        testWidgets('poker $mode $bots $tag', (tester) async {
          tester.view.physicalSize = size;
          tester.view.devicePixelRatio = 1.0;
          final game = GameProvider(p2pService: P2PService())
            ..setHost(true)
            ..setMultiplayer(false)
            ..setGameMode(mode)
            ..setBotCount(bots);
          final key = GlobalKey();
          await tester.pumpWidget(ChangeNotifierProvider.value(value: game, child: _app(key, const GameTableScreen())));
          await _settle(tester);
          await _snap(tester, key, 'poker_${mode.name}_${bots}_$tag');
          await _teardown(tester);
        });
      }
    }

    for (final bots in [1, 3, 5]) {
      testWidgets('blackjack $bots $tag', (tester) async {
        tester.view.physicalSize = size;
        tester.view.devicePixelRatio = 1.0;
        final bj = BlackjackProvider()..initSinglePlayer(botCount: bots);
        final key = GlobalKey();
        await tester.pumpWidget(ChangeNotifierProvider.value(value: bj, child: _app(key, const BlackjackTableScreen())));
        bj.placeBet('0', 100);
        await _settle(tester);
        await _snap(tester, key, 'blackjack_${bots}_$tag');
        await _teardown(tester);
      });
    }
  }
}
